\section{Problem definition and motivation}

\begin{frame}{Problem definition}
Given a regular triangulation \(\mathcal{T}\) of a polygonal bounded Lipschitz domain \(\Omega \subseteq \mathbb{R}^2 \) and \( f \in L^2(\Omega)\), the discretized weak form of the PMP seeks \(\uCR \in \mathrm{CR}^1_0 (\mathcal{T})\) s.t.
	\[
    \int_\Omega \nabla_\mathrm{pw} \uCR  \nabla_\mathrm{pw} \vCR \dx = \int_\Omega f \vCR \dx \quad \text{for all } \vCR \in \mathrm{CR}^1_0(\mathcal{T})
    \]
\end{frame}

\begin{frame}{Motivation}
What about \(f \notin L^2(\Omega)\)?

\begin{itemize}
    \item point charges or concentrated forces
    \item entry point for other, more complex PDEs
    \item limitations of the Finite Element Method
\end{itemize}
 
\begin{block}{}
Therefore we consider the new problem
    \[
    \int_\Omega \nabla_\mathrm{pw} u_\text{CR} \nabla_\mathrm{pw} v_\text{CR} \dx = \int_\Omega f v_\text{C} \dx
    \]
with \(\vC := J \vCR \) the conforming companion
\end{block}
\end{frame}

\subsection{J1 Operator}
\begin{frame}[fragile]{J1 Operator}

\begin{columns}[T]
% ---------------- LEFT COLUMN ----------------
\begin{column}{0.48\textwidth}
\begin{block}{Definition}
Define \(
J_1 : \mathrm{CR}^1_0(\mathcal{T}) \to S^1_0(\mathcal{T})\)
by
\[
|\mathcal{T}(z)|^{-1}
\sum_{T \in \mathcal{T}(z)} {\vCR|_T(z)}
\]
for all \( z \in \mathcal{N}(\Omega) \), plus boundary conditions.
\end{block}
\end{column}

\subsection{J2 Operator}
\begin{frame}[fragile]{J2 Operator}

\begin{columns}[T]
% ---------------- LEFT COLUMN ----------------
\begin{column}{0.48\textwidth}
\begin{block}{Definition}
Define \(
J_2 : S^1_0(\mathcal{T}) \to S^2_0(\mathcal{T})\)
as
\[
J_2 \vCR := J_1 \vCR + \sum_{E \in \mathcal{E}(\Omega)} \mathrm{b_E}\flat_E
\]
where \[\mathrm{b_E} := \frac{1}{|E|}\int_{E} \left( v_{CR} - J_{1} v_{CR} \right)\, \mathrm{d}s\] and \[\flat_E|_T := 6 \prod_{P \in \mathcal{N}(E)} \lambda_P \]
\end{block}
\end{column}

% ---------------- RIGHT COLUMN ----------------
\begin{column}{0.48\textwidth}

\begin{block}{ MATLAB Code}
\footnotesize
\begin{verbatim}
J2 = vCR-sum(J1(n4s),2)/2;
\end{verbatim}
\end{block}
\end{column}
\end{columns}
\end{frame}
\subsection{J3 Operator}
\begin{frame}[fragile]{J3 Operator}
\begin{block}{Definition}
Define \(
J_3 : \mathrm{CR}^1_0(\mathcal{T}) \to S^4_0(\mathcal{T})\)
\[
J_3 \vCR := J_2 \vCR
+ \sum_{T \in \mathcal{T}} \sum_{P \in \mathcal{N}(T)}
\left(
  \int_T \bigl( \vCR - J_2 \vCR \bigr)\, \Phi_{T,P} \dx
\right)
\Phi_{T,P} b_T
\]
where \(\Phi_{T,P} := |T|^{-1/2} (\alpha - \beta \lambda_P)\) and \(b_T := 27\prod_{P \in \mathcal{N}(T)} \lambda_P \)
\end{block}
\end{frame}

\begin{frame}{expected convergence rates}
\begin{itemize}
    \item \(\sum_{E \in \mathcal{E}(\gamma)} |E| \norm{\ell}^2_{L^2(E)}  = |E| \norm{\ell}^2_{L^2(\gamma)} \approx \mathrm{ndof}^{-1/2}\)
    \vspace{1.2em}
    \item for the normal PMP: \(\sum_{T \in \mathcal{T}} |T| \sum_{E \in \mathcal{E}(T)} \norm{[\partial u/\partial s]_E}^2_{L^2(E)} \approx \mathrm{ndof}^{-1} \)
    \vspace{1.2em}
    \item Therefore we expect a convergence rate of \(-\frac{1}{4}\) (at best)
\end{itemize}
\end{frame}

\section{Conclusion and outlook}
\begin{frame}{Conclusion}
\begin{itemize}
    \item it works! and it looks cool :)
    \item observed convergence rate matches expectations (\(s = -1/4\))
    \item convergence rate is independent of lineload \(\ell \in L^2(\gamma)\)
\end{itemize}  
\end{frame}
\begin{frame}{Outlook}
\begin{itemize}
    \item using data structures and local nature of FEM to enable the computation up to ... degrees of freedom
    \item using adaptive mesh refinement
    \item implement for general lines \(\gamma \subseteq \Omega\)
    \item implement for general \(f = f_\mathrm{vol} + f_\mathrm{line}\)
    \item solve for (\(\Omega, f\)) where exact solution \(u\) is known
\end{itemize}
\end{frame}
